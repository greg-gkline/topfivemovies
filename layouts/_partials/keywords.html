{{- /* The keywords of one item: its title and the people in it (a review's cast, a top-five list's entries). Index pages join the keywords of their newest items. */ -}} {{- $words := slice .Title -}} {{- with .Params.starring -}} {{- $words = $words | append (replace . " and " " ") -}} {{- end -}} {{- range slice "item_1" "item_2" "item_3" "item_4" "item_5" -}} {{- with index $.Params . -}} {{- $words = $words | append (.title | $.RenderString | plainify | htmlUnescape) -}} {{- end -}} {{- end -}} {{- return $words -}}
